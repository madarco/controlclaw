/** Atomically closes new suspension admission before synchronous inspection. */
function tryBeginGatewaySuspendAdmission(e, t) {
	if (isGatewayWorkAdmissionClosed()) return null;
	GATEWAY_WORK_ADMISSION_STATE.suspendPhase = "preparing";
	const n = ++GATEWAY_WORK_ADMISSION_STATE.suspendGeneration;
	GATEWAY_WORK_ADMISSION_STATE.suspendInvalidated = e;
	const r = JSON.stringify(t ?? {
		owner: "internal",
		leaseId: String(n),
		caller: "in-process",
		reason: "suspend"
	});
	GATEWAY_WORK_ADMISSION_STATE.suspendAttribution = r;
	admissionLog.info(`admission closed: suspend phase ${r}`);
	notifyGatewaySuspendAdmission();
	const transition = (e, t, i) => {
		if (GATEWAY_WORK_ADMISSION_STATE.suspendGeneration !== n || GATEWAY_WORK_ADMISSION_STATE.suspendPhase !== e) return false;
		GATEWAY_WORK_ADMISSION_STATE.suspendPhase = t;
		if (t === "accepting") {
			GATEWAY_WORK_ADMISSION_STATE.suspendInvalidated = void 0;
			resolveSuspendOpenWaiters();
			GATEWAY_WORK_ADMISSION_STATE.suspendAttribution = void 0;
			admissionLog.info(`admission reopened: suspend phase ${r} exit=${i}`);
		}
		notifyGatewaySuspendAdmission();
		return true;
	};
	return {
		drain: () => transition("preparing", "draining"),
		commit: () => transition("preparing", "prepared") || transition("draining", "prepared"),
		rollback: (e = "rollback") => transition("preparing", "accepting", e),
		release: (e = "released") => transition("draining", "accepting", e) || transition("prepared", "accepting", e)
	};
}
/** Clears restart/suspend admission during SIGUSR2 and isolated tests. */
function resetGatewayWorkAdmission() {
	GATEWAY_WORK_ADMISSION_STATE.restartDrainController.abort(new GatewayDrainingError("gateway runtime reset"));
	for (const e of GATEWAY_WORK_ADMISSION_STATE.activeRootWork) {
		e.references = 0;
		e.retiredByReset = true;
		e.released = true;
		notifyGatewayRootWorkRelease(e, "reset");
	}
	GATEWAY_WORK_ADMISSION_STATE.activeRootWork.clear();
	GATEWAY_WORK_ADMISSION_STATE.restartDrainReason = void 0;
	GATEWAY_WORK_ADMISSION_STATE.restartDrainController = new AbortController();
	GATEWAY_WORK_ADMISSION_STATE.shutdownCleanupController.abort();
	GATEWAY_WORK_ADMISSION_STATE.shutdownCleanupController = createShutdownCleanupController();
	GATEWAY_WORK_ADMISSION_STATE.restartSignalPending = false;
	GATEWAY_WORK_ADMISSION_STATE.restartSignalGeneration += 1;
	if (GATEWAY_WORK_ADMISSION_STATE.suspendPhase !== "accepting") invalidateSuspendAdmission();
	else {
		GATEWAY_WORK_ADMISSION_STATE.suspendGeneration += 1;
		GATEWAY_WORK_ADMISSION_STATE.suspendInvalidated = void 0;
	}
	resolveSuspendOpenWaiters();
}
